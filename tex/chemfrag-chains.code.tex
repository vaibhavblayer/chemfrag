%% Geometry
%% ---------------------------------------------------------------------------
%% A chain of n atoms starts on the current atom (C1) and draws n-1 bonds
%% that zigzag around the incoming direction:  +30, -30, +30, ... degrees
%% (s = +1, "start=up") or the mirror image (s = -1, "start=down").
%% All angles are relative ("[::a]"), so the chain follows whatever bond
%% leads into it.
%%   bond 1           : [::30s]
%%   bond j (j >= 2)  : [::60s(-1)^(j+1)]
%% Substituents point away from the chain:
%%   C1 (molecule start) : [::180-30s]   (where the previous zig would be)
%%   C1 (after a bond)   : [::-90s]
%%   Ck (1 < k < n)      : [::60s(-1)^k]
%%   Cn                  : [::-60s(-1)^n] (continues the zigzag)
%% At the end a tiny invisible bond turns towards the next zig, so
%% "-[::0]" continues the zigzag naturally.
%% ---------------------------------------------------------------------------

\cs_new:Npn \chemfragchain #1
  { \exp:w \exp_after:wN \exp_end: \expanded { \__chemfrag_chain:Vn \l__chemfrag_defaults_tl {#1} } }

\cs_new:Npn \__chemfrag_chain:nn #1#2
  { \__chemfrag_chain_main:n { #2 , #1 } }
\cs_generate_variant:Nn \__chemfrag_chain:nn { V }

%% {n} {s} {standalone}
\cs_new:Npn \__chemfrag_chain_main:n #1
  {
    \exp_args:Ne \__chemfrag_chain_main_a:nn
      {
        { \__chemfrag_chain_length:n {#1} }
        {
          \exp_args:Ne \str_if_eq:nnTF
            { \__chemfrag_get:nnn {#1} { start } { up } } { down }
            { -1 } { 1 }
        }
        { \__chemfrag_if_standalone:TF { 1 } { 0 } }
      }
      {#1}
  }
\cs_new:Npn \__chemfrag_chain_main_a:nn #1#2
  { \__chemfrag_chain_emit:nnnn #1 {#2} }

%% Length: "n=<int>" or a bare integer; default 3.
\cs_new:Npn \__chemfrag_chain_length:n #1
  {
    \exp_args:Ne \__chemfrag_chain_length_aux:nn
      { \exp_args:Ne \__chemfrag_get_aux:nn { \__chemfrag_bare_int:n {#1} } { 3 } }
      {#1}
  }
\cs_new:Npn \__chemfrag_chain_length_aux:nn #1#2
  { \__chemfrag_int:nnn {#2} { n } {#1} }

%%   #1 = n, #2 = s, #3 = standalone, #4 = keys
\cs_new:Npn \__chemfrag_chain_emit:nnnn #1#2#3#4
  {
    \__chemfrag_name:nn {#4} { 1 }
    \__chemfrag_atom:nn {#4} { 1 }
    \__chemfrag_branch:nnn {#4} { 1 }
      {
        \int_compare:nNnTF {#1} = { 1 }
          { \int_eval:n { 180 - 30 * (#2) } }
          {
            \int_compare:nNnTF {#3} = { 1 }
              { \int_eval:n { 180 - 30 * (#2) } }
              { \int_eval:n { -90 * (#2) } }
          }
      }
    \__chemfrag_chain_loop:nnnn { 2 } {#1} {#2} {#4}
    % final turn so that "-[::0]" continues the zigzag
    \exp_args:Ne \str_if_eq:nnF { \__chemfrag_get:nnn {#4} { exit } { end } } { none }
      {
        \__chemfrag_jump:nn
          { \int_eval:n { -60 * (#2) * \int_if_odd:nTF {#1} { -1 } { 1 } } }
          { 0.01 }
        \__chemfrag_phantom:nn {#4} {#1}
      }
  }

%%   #1 = k, #2 = n, #3 = s, #4 = keys
\cs_new:Npn \__chemfrag_chain_loop:nnnn #1#2#3#4
  {
    \int_compare:nNnF {#1} > {#2}
      {
        % bond k-1
        \__chemfrag_bond:nnnn
          { \exp_args:Nne \__chemfrag_chain_bond_char:nn {#4} { \int_eval:n { #1 - 1 } } }
          {
            \int_compare:nNnTF {#1} = { 2 }
              { \int_eval:n { 30 * (#3) } }
              { \int_eval:n { 60 * (#3) * \int_if_odd:nTF {#1} { -1 } { 1 } } }
          }
          { } { }
        % atom k
        \__chemfrag_name:nn {#4} {#1}
        \__chemfrag_atom:nn {#4} {#1}
        \__chemfrag_branch:nnn {#4} {#1}
          {
            \int_compare:nNnTF {#1} = {#2}
              { \int_eval:n { -60 * (#3) * \int_if_odd:nTF {#2} { -1 } { 1 } } }
              { \int_eval:n {  60 * (#3) * \int_if_odd:nTF {#1} { -1 } { 1 } } }
          }
        \exp_args:Ne \__chemfrag_chain_loop:nnnn
          { \int_eval:n { #1 + 1 } } {#2} {#3} {#4}
      }
  }

%% Bond j:  double={...}  triple={...}  bonds={<pattern>}  (single otherwise)
\cs_new:Npn \__chemfrag_chain_bond_char:nn #1#2
  {
    \__chemfrag_if_listed:nnnTF {#1} { triple } {#2}
      { \c_tilde_str }
      {
        \__chemfrag_if_listed:nnnTF {#1} { double } {#2}
          { = }
          {
            \exp_args:Ne \__chemfrag_bond_char:nn
              { \__chemfrag_get:nnn {#1} { bonds } { } } {#2}
          }
      }
  }

%% !{chain}{<keys>}
\__chemfrag_submol_arg:nn { chain } { !\chemfragchain }
